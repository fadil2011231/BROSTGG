// Fetch Minecraft Username API (Minotar) $("fetch-btn").onclick = () => { const user = $("username-input").value.trim(); if(!user) { notify("Please enter a username!"); return; } notify("Fetching skin for " + user + "..."); const img = new Image(); // Custom check for BROSTGG if (user.toUpperCase() === "BROSTGG") { // Local file loading - crossOrigin is removed here to prevent file:// protocol block img.src = "image_933a9c.png"; } else { // Web API loading - requires crossOrigin img.crossOrigin = "Anonymous"; img.src = `https://minotar.net/skin/${user}`; } img.onload = () => { loadIntoEditor(img, user + "-Skin", true); notify(user + "'s skin loaded successfully!"); }; img.onerror = () => { notify("Could not find a skin for that username."); }; };
